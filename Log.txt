    def show_output_video(self):
        video_path = os.path.join(self.output_dir, "output_video.avi")
        if not os.path.exists(video_path):
            messagebox.showerror("Error", "Output video not found!")
            return
        
        def play_video():
            cap = cv2.VideoCapture(video_path)
            while cap.isOpened():
                ret, frame = cap.read()
                if not ret:
                    break
                frame = cv2.resize(frame, (640, 360))
                rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                img = ImageTk.PhotoImage(Image.fromarray(rgb))
                self.output_display.config(image=img)
                self.output_display.image = img
                self.output_display.update()
                time.sleep(0.03)
            cap.release()
        
        threading.Thread(target=play_video, daemon=True).start()

    def show_heatmap(self):
        heatmap_path = os.path.join(self.output_dir, "heatmap_overlay.jpg")
        if not os.path.exists(heatmap_path):
            messagebox.showerror("Error", "Heatmap not found!")
            return
        
        img = cv2.imread(heatmap_path)
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
        img = ImageTk.PhotoImage(Image.fromarray(img))
        self.output_display.config(image=img)
        self.output_display.image = img
